cmake_minimum_required(VERSION 3.31)
project(open_meteo_consumer LANGUAGES CXX)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -DOPEN_METEO_PARENT_FLAGS")
if(OPEN_METEO_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(open_meteo_cpp SOURCE_DIR "${OPEN_METEO_SOURCE_DIR}")
    FetchContent_MakeAvailable(open_meteo_cpp)
    if(OPEN_METEO_BUILD_TESTS OR OPEN_METEO_BUILD_EXAMPLES OR OPEN_METEO_ENABLE_LTO)
        message(FATAL_ERROR "SDK subproject defaults must be off")
    endif()
    get_directory_property(sdk_release_flags DIRECTORY "${OPEN_METEO_SOURCE_DIR}"
        DEFINITION CMAKE_CXX_FLAGS_RELEASE)
    if(NOT sdk_release_flags STREQUAL "-O2 -DNDEBUG -DOPEN_METEO_PARENT_FLAGS" OR
       NOT CMAKE_CXX_FLAGS_RELEASE STREQUAL "-O2 -DNDEBUG -DOPEN_METEO_PARENT_FLAGS")
        message(FATAL_ERROR "SDK overwrote the parent's release flags")
    endif()
else()
    find_package(open_meteo ${OPEN_METEO_EXPECTED_VERSION} EXACT CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE open_meteo::open_meteo)

if(README_EXAMPLE_PATH)
    add_executable(readme_example "${README_EXAMPLE_PATH}")
    target_link_libraries(readme_example PRIVATE open_meteo::open_meteo)
endif()
